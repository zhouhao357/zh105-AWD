一、条件语句

if / else if / else、switch、三元表达式

1. if / else if / else —— 成绩等级判定

条件 
function getGrade(score) {
    if (score >= 90)      return '优秀 A';
    else if (score >= 80) return '良好 B';
    else if (score >= 60) return '及格 C';
    else                    return '不及格 D';
}

3. 三元运算符 —— 奇偶判断

条件

function isOdd(n) {
    return n % 2 === 0 ? '偶数' : '奇数';

二、循环语句

for、while、do-while、for...of / forEach

1. for 循环 —— 求 1~N 的和

循环 
function sumToN(n) {
    let sum = 0;
    for (let i = 1; i <= n; i++) {
        sum += i;
    }
    return sum;
}
输出

2. while 循环 —— 猜数字（二分逼近）

循环 
function guessNumber(target) {
    let low = 1, high = 100, steps = 0;
    while (low <= high) {
        let mid = Math.floor((low + high) / 2);
        steps++;
        if (mid === target) return `猜到 ${mid}，用了 ${steps} 次`;
        else if (mid < target) low = mid + 1;
        else high = mid - 1;
    }
}
输出

3. do-while 与 for...of —— 遍历数组

循环 
const fruits = ['苹果', '香蕉', '橘子'];
let i = 0;
do {
    console.log(fruits[i]);
    i++;
} while (i < fruits.length);

// 更简洁的 for...of
for (const f of fruits) {
    console.log(f);
}
输出

三、综合示例 —— 循环 + 条件

打印 1~20：逢 7 过（参考 15-1），其余按奇偶标记

完整代码

综合 
for (let i = 1; i <= 20; i++) {
    if (i % 7 === 0 || String(i).includes('7')) {
        console.log(i + ': 过（逢七必过）');
    } else if (i % 2 === 0) {
        console.log(i + ': 偶数');
    } else {
        console.log(i + ': 奇数');
    }
}
输出
