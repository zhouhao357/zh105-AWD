## 四、核心代码 3：页面切换（data-page 驱动，侧边栏与底栏联动）

```js
/* 入口：<a class="zh-side-link is-active" data-page="home"> / <div class="zh-nav-item" data-page="search"> */
function zhSwitchPage(name) {
    var target = $('.zh-page[data-page="' + name + '"]');
    if (!target) return;
    $$('.zh-page').forEach(function (p) { p.classList.remove('is-active'); });
    target.classList.add('is-active');
    $$('[data-page]').forEach(function (el) {
        el.classList.toggle('is-active', el.getAttribute('data-page') === name);
    });
    window.scrollTo(0, 0);
    zhCloseDrawer();
}
$$('[data-page]').forEach(function (el) {
    el.addEventListener('click', function () {
        zhSwitchPage(el.getAttribute('data-page'));
    });
});
```

## 五、核心代码 4：轮播图（自动播放 + 箭头 + 指示点 + 悬停暂停）

```js
/* 结构：.zh-carousel > .zh-slide(3) + .zh-arrow-prev/.zh-arrow-next + .zh-dot(3) */
var zhSlides = $$('.zh-slide');
var zhDots = $$('.zh-dot');
var zhCur = 0, zhTimer = null;

function zhShowSlide(i) {
    zhSlides[zhCur].classList.remove('is-active');
    zhDots[zhCur].classList.remove('is-active');
    zhCur = (i + zhSlides.length) % zhSlides.length;
    zhSlides[zhCur].classList.add('is-active');
    zhDots[zhCur].classList.add('is-active');
}
function zhAutoPlay() {
    clearInterval(zhTimer);
    zhTimer = setInterval(function () { zhShowSlide(zhCur + 1); }, 4000);
}
$('.zh-arrow-prev').addEventListener('click', function () { zhShowSlide(zhCur - 1); zhAutoPlay(); });
$('.zh-arrow-next').addEventListener('click', function () { zhShowSlide(zhCur + 1); zhAutoPlay(); });
zhDots.forEach(function (d, i) {
    d.addEventListener('click', function () { zhShowSlide(i); zhAutoPlay(); });
});
$('.zh-carousel').addEventListener('mouseenter', function () { clearInterval(zhTimer); });
$('.zh-carousel').addEventListener('mouseleave', zhAutoPlay);
zhAutoPlay();
```

## 六、核心代码 5：播放器（点歌即播、进度拖拽、切歌、空格控制）

```js
/* 歌曲数据集中在脚本顶部，与列表渲染共用同一份数据源 */
var ZH_SONGS = [
    { name: '晴天', artist: '周杰伦', album: '叶惠美', dur: 269, img: 0 },
    { name: '稻香', artist: '周杰伦', album: '魔杰座', dur: 236, img: 1 }
    /* ... 共 8 首 ... */
];

var zhPlayer = { index: 0, playing: false, pos: 0, timer: null };

function zhLoadSong(i, play) {
    zhPlayer.index = (i + ZH_SONGS.length) % ZH_SONGS.length;
    var song = ZH_SONGS[zhPlayer.index];
    $('.zh-pb-name').textContent = song.name;
    $('.zh-pb-artist').textContent = song.artist;
    $('.zh-pb-cover').style.backgroundImage = 'url(' + ZH_IMG[song.img % ZH_IMG.length] + ')';
    zhPlayer.pos = 0;
    $$('.zh-song').forEach(function (el, k) {
        el.classList.toggle('is-playing', k === zhPlayer.index);   /* 当前播放高亮 */
    });
    zhRenderProgress();
    if (play !== false) zhSetPlaying(true);
}

function zhSetPlaying(on) {
    zhPlayer.playing = on;
    $('.zh-pb-btn-main').classList.toggle('is-playing', on);
    $('.zh-player').classList.toggle('is-playing', on);   /* 唱片旋转动画 */
    clearInterval(zhPlayer.timer);
    if (on) {
        zhPlayer.timer = setInterval(function () {
            zhPlayer.pos += 1;
            if (zhPlayer.pos >= ZH_SONGS[zhPlayer.index].dur) zhLoadSong(zhPlayer.index + 1, true);
            else zhRenderProgress();
        }, 1000);
    }
}

/* 进度条：pointer 事件实现点击定位与拖拽 */
zhBar.addEventListener('pointerdown', function (e) {
    zhBar.setPointerCapture(e.pointerId);
    zhSeek(e);
});
zhBar.addEventListener('pointermove', function (e) {
    if (zhBar.hasPointerCapture(e.pointerId)) zhSeek(e);
});

/* 空格键播放 / 暂停 */
document.addEventListener('keydown', function (e) {
    if (e.code === 'Space' && !/INPUT|TEXTAREA/.test(document.activeElement.tagName)) {
        e.preventDefault();
        zhSetPlaying(!zhPlayer.playing);
    }
});
```

## 七、核心代码 6：搜索联动（顶栏回车跳转 + 实时过滤）

```js
function zhFilter(kw) {
    kw = kw.trim().toLowerCase();
    var n = 0;
    $$('.zh-hot-item').forEach(function (li) {
        var hit = !kw || li.textContent.toLowerCase().indexOf(kw) > -1;
        li.style.display = hit ? '' : 'none';
        if (hit) n++;
    });
    $('.zh-empty').hidden = n > 0;      /* 无结果时显示提示 */
}

/* 顶栏搜索框回车 → 跳转搜索页并同步关键词 */
$('.zh-search-input').addEventListener('keydown', function (e) {
    if (e.key !== 'Enter') return;
    var kw = e.currentTarget.value.trim();
    if (!kw) return;
    zhSwitchPage('search');
    $('.zh-search-box-input').value = kw;
    zhFilter(kw);
});
$('.zh-search-box-input').addEventListener('input', function (e) { zhFilter(e.currentTarget.value); });
```
